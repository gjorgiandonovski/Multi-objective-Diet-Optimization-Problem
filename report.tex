\begin{frame}
\frametitle{Problem Definition}

\textbf{Goal.} Design 7-day meal plans that minimize two conflicting objectives while respecting food category constraints.

\vspace{0.3cm}

\textbf{Problem structure.}
\begin{itemize}
  \item $N=2{,}616$ food items with nutritional data (kcal, carbs, fat, protein).
  \item 5 user profiles (ages 17, 30, 40, 55, 72) with targets 1{,}401--3{,}103 kcal/day.
  \item 77 decision variables: 7 days $\times$ 11 meal slots per day.
  \item Each slot restricted to admissible set $D_j$ (snacks, breakfast, lunch, dinner).
\end{itemize}

\vspace{0.3cm}

\textbf{Two objectives (minimize).}
\begin{align*}
f_1(\mathbf{x}) &= \sum_{d=1}^{7} \left|K(\mathbf{x},d) - T\right|
\quad \text{(caloric deviation)} \\
f_2(\mathbf{x}) &= \sum_{d=1}^{7} \Bigl(\left|p_C-50\right| + \left|p_F-27.5\right| + \left|p_P-22.5\right|\Bigr)
\quad \text{(macronutrient imbalance)}
\end{align*}

\textbf{Feasibility.} $x_j \in D_j$ for all $j=1,\dots,77$ (enforced by encoding).

\end{frame}

%================================================
\begin{frame}
\frametitle{Solution Encodings}

\textbf{Two representation types.}

\begin{itemize}
  \item \textbf{Direct-Index (integer vector):}
  \[
  \mathbf{x} \in \{1,\dots,2616\}^{77}.
  \]
  Gene $j$ stores the food ID assigned to slot $j$.
  Feasibility is enforced by sampling and mutating within $D_j$.

  \item \textbf{Random-Key (real vector):}
  \[
  \mathbf{k} \in [0,1]^{77}.
  \]
  Decoding: interval $[0,1]$ split into $|D_j|$ equal bins; $k_j$ selects the corresponding food.
  Any key vector decodes to a feasible plan.
\end{itemize}

\vspace{0.3cm}

\textbf{Feasibility-preserving design.}
\begin{itemize}
  \item Generator only draws from admissible domains $D_j$.
  \item All variators (mutation, crossover) respect domain constraints.
  \item Zero infeasible solutions are generated; no repair/penalty strategies are needed.
\end{itemize}

\end{frame}

%================================================
\begin{frame}
\frametitle{Fitness Function Computation}

\textbf{Input:} solution $\mathbf{x}$ (or decoded $\mathbf{x}$ from random keys).

\vspace{0.2cm}

\textbf{For each day $d \in \{1,\dots,7\}$:}
\begin{enumerate}
  \item Sum daily energy and macronutrients from assigned food items.
  \item Compute daily caloric total $K(\mathbf{x},d)$.
  \item Compute macronutrient energy percentages $p_C, p_F, p_P$ (Atwater factors: 4/9/4).
  \item Accumulate deviations: $\left|K(\mathbf{x},d) - T\right|$ and $\left|p_M - \text{target}_M\right|$.
\end{enumerate}

\vspace{0.2cm}

\textbf{Final objectives:}
\begin{itemize}
  \item $f_1 = \sum_{d=1}^{7} \left|K(\mathbf{x},d) - T\right|$.
  \item $f_2 = \sum_{d=1}^{7}\left(\left|p_C-50\right| + \left|p_F-27.5\right| + \left|p_P-22.5\right|\right)$.
\end{itemize}

\vspace{0.2cm}

\textbf{Multi-objective fitness.} Both $f_1$ and $f_2$ are minimized; Pareto dominance ranks solutions.

\end{frame}

%================================================
\begin{frame}
\frametitle{Four Metaheuristics}

\textbf{Evolutionary Algorithms}
\begin{itemize}
  \item \textbf{NSGA-II:} non-dominated sorting + crowding distance; pop=80, gen=80; evaluated with both encodings.
  \item \textbf{PAES:} $(1{+}1)$ evolution strategy with adaptive grid archive; gen=800; direct-index only.
\end{itemize}

\textbf{Swarm Algorithms}
\begin{itemize}
  \item \textbf{MOPSO:} particles guided by sigma-rule leaders from external archive; pop=30, gen=80; random-key only.
  \item \textbf{P-ACO:} ants construct plans via per-slot pheromone sampling; archive-based reinforcement; pop=60, gen=40; direct-index only.
\end{itemize}

\vspace{0.3cm}

\textbf{All share:} same objectives, same feasibility logic, and the same metric/statistical pipeline for fair comparison.

\end{frame}

%================================================
\begin{frame}
\frametitle{Variation Operators and Search Behavior}

\textbf{Direct-Index operators (NSGA-II, PAES, P-ACO construction output).}
\begin{itemize}
  \item \textbf{Mutation:} if gene $j$ is selected, replace $x_j$ with a fresh uniform draw from $D_j$.
  \item \textbf{Crossover:} multi-point crossover; offspring remain feasible because genes are copied position-wise from feasible parents.
\end{itemize}

\vspace{0.25cm}

\textbf{Random-Key operators (NSGA-II-RK, MOPSO).}
\begin{itemize}
  \item \textbf{Mutation:} Gaussian perturbation ($\sigma=0.1$) + reflection/clamp to $[0,1]$.
  \item \textbf{Crossover:} multi-point crossover in key space.
\end{itemize}

\vspace{0.25cm}

\textbf{Behavioral contrast.}
\begin{itemize}
  \item NSGA-II explores the trade-off surface with diversity pressure.
  \item PAES follows a single trajectory and accepts moves that improve archive coverage.
\end{itemize}

\end{frame}

%================================================
\begin{frame}
\frametitle{MOPSO and P-ACO: Swarm Implementations}

\textbf{MOPSO (random-key space).}
\begin{itemize}
  \item Particle position $\mathbf{k}_i(t)\in[0,1]^{77}$; velocity $\mathbf{v}_i(t)\in\mathbb{R}^{77}$.
  \item External archive stores non-dominated solutions; pruned by crowding distance.
  \item Leader selection favors sparse regions (sigma-rule / diversity-based leader pick).
  \item Decoding maps keys $\mathbf{k}$ to a feasible food plan $\mathbf{x}$ after each update.
\end{itemize}

\vspace{0.3cm}

\textbf{P-ACO (direct-index space).}
\begin{itemize}
  \item Ants sequentially fill the 77 slots by sampling from $D_j$ with pheromone weights $\tau_{j,f}$.
  \item Pheromone reinforcement uses archive members (diversity-weighted deposits).
  \item Evaporation prevents trail collapse; bounds $\tau_{\min},\tau_{\max}$ maintain exploration.
\end{itemize}

\end{frame}

%================================================
\begin{frame}
\frametitle{Experimental Framework}

\textbf{Hyperparameter tuning.}
\begin{itemize}
  \item Performed on 2 subjects with 5 replicates per parameter cell.
  \item Select by highest mean hypervolume (tie-break by runtime).
\end{itemize}

\vspace{0.3cm}

\textbf{Full benchmark.}
\begin{itemize}
  \item 5 configurations: NSGA-II (DI, RK), PAES (DI), MOPSO (RK), P-ACO (DI).
  \item 5 subjects $\times$ 30 replicates per configuration $\Rightarrow$ 750 total runs.
  \item Paired seeds across configurations for paired tests.
\end{itemize}

\vspace{0.3cm}

\textbf{Metrics and statistics.}
\begin{itemize}
  \item HV, IGD, spacing, $\Delta$-spread, best $f_1$, best $f_2$, runtime.
  \item Friedman aligned ranks (omnibus) + Wilcoxon signed-ranks (pairwise) with correction.
\end{itemize}

\end{frame}

\textbf{Key findings.}

  \item NSGA-II dominates Pareto coverage (HV, IGD).

\begin{frame}
\frametitle{Statistical Significance (Paired Replicates)}

\textbf{Friedman aligned ranks (omnibus).}
\begin{itemize}
  \item Rejects null across configurations on all metrics ($p < 10^{-70}$).
\end{itemize}

\vspace{0.25cm}

\textbf{Wilcoxon signed-ranks + multiple-comparison correction.}
\begin{itemize}
  \item NSGA-II (both variants) significantly better than PAES/MOPSO/P-ACO on HV.
  \item NSGA-II encodings: HV difference not significant ($p=0.359$).
  \item Non-NSGA-II methods: not significantly different from one another on HV after correction.
\end{itemize}

\vspace{0.25cm}

\textbf{Conclusion.} NSGA-II superiority is statistically supported; encoding choice is secondary.

\end{frame}

\begin{frame}
\frametitle{Implementation Insights}

\textbf{Feasibility-preserving encoding.}
\begin{itemize}
  \item All solutions satisfy per-slot category constraints by construction ($x_j\in D_j$).
  \item No repair/penalty/death-penalty overhead is needed in the main benchmark.
\end{itemize}

\vspace{0.25cm}

\textbf{Algorithm performance drivers.}
\begin{itemize}
  \item \textbf{NSGA-II:} best overall Pareto coverage via sorting + crowding distance.
  \item \textbf{PAES:} archive-driven exploitation; strong low-$f_2$ solutions.
  \item \textbf{MOPSO:} fast continuous dynamics; weaker precision through decoding.
  \item \textbf{P-ACO:} pheromone learning from archive; competitive but computationally heavy in Python.
\end{itemize}

\vspace{0.25cm}

\textbf{Takeaway.} The metaheuristic family dominates performance differences more than the encoding choice.

\end{frame}

%================================================
\begin{frame}
\frametitle{Conclusion}

\begin{itemize}
  \item Best overall: NSGA-II (both encodings) for Pareto front coverage.
  \item Use PAES when the main goal is a very low macronutrient deviation solution (low $f_2$).
  \item Use MOPSO when runtime is the primary constraint.
  \item Feasibility-preserving representations simplify optimization and make constraint-handling unnecessary.
  \item Statistical tests confirm the observed differences are not due to stochastic noise.
\end{itemize}

\end{frame}
